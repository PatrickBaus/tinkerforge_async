26async def join(*generators: AsyncGenerator[T, None]) -> AsyncGenerator[T, None]: 

27 """ 

28 Merge the items of all `generators` into a single stream, in the order they arrive. The stream ends when all 

29 generators are exhausted. If one generator raises an exception, it is raised by the stream, and all generators are 

30 closed. Leaving the loop early (e.g. with `break`) closes all generators, too. 

31 

32 Note: This is not implemented with a `TaskGroup`, because task groups must not be used in async generators. They 

33 can not handle the `GeneratorExit` exception, that is thrown into the generator when it is closed. 

34 """ 

35 done = object() # Every producer puts this sentinel into the queue when it is finished 

36 queue: asyncio.Queue[object] = asyncio.Queue(maxsize=1) 

37 

38 async def pump(generator: AsyncGenerator[T, None]) -> None: 

39 try: 

40 async with aclosing(generator): 

41 async for item in generator: 

42 await queue.put(item) 

43 except Exception as exc: # pylint: disable=broad-exception-caught # Forwarded to the consumer 

44 await queue.put(_Failure(exc)) 

45 else: 

46 await queue.put(done) 

47 

48 producers = [asyncio.create_task(pump(generator)) for generator in generators] 

49 try: 

50 remaining = len(producers) 

51 while remaining: 

52 item = await queue.get() 

53 if item is done: 

54 remaining -= 1 

55 elif isinstance(item, _Failure): 

56 raise item.exception 

57 else: 

58 yield cast(T, item) 

59 finally: 

60 for producer in producers: 

61 producer.cancel() 

62 await asyncio.gather(*producers, return_exceptions=True)

65def seconds_to_ms(value: float | Decimal) -> int: 

66 """ 

67 Convert a duration in s to the integer value in ms used by the Tinkerforge protocol. Rounding avoids floating point 

68 errors, e.g. 0.29 s * 1000 = 290.00000000000006. 

69 """ 

70 return round(Decimal(str(value)) * 1000)
