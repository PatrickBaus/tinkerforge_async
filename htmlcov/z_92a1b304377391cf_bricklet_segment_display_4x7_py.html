76 async def set_segments( 

77 self, 

78 segments: tuple[int, int, int, int] = (0, 0, 0, 0), 

79 brightness: int = 7, 

80 colon: bool = False, 

81 response_expected: bool = True, 

82 ) -> None: 

83 """ 

84 The 7-segment display can be set with bitmaps. Every bit controls one 

85 segment: 

86 

87 .. image:: /Images/Bricklets/bricklet_segment_display_4x7_bit_order.png 

88 :scale: 100 % 

89 :alt: Bit order of one segment 

90 :align: center 

91 

92 For example to set a "5" you would want to activate segments 0, 2, 3, 5 and 6. 

93 This is represented by the number 0b01101101 = 0x6d = 109. 

94 

95 The brightness can be set between 0 (dark) and 7 (bright). The colon 

96 parameter turns the colon of the display on or off. 

97 """ 

98 if not all(0 <= segment <= 127 for segment in segments): 

99 raise ValueError( 

100 "Invalid argument: expected all(0 <= segment <= 127 for segment in segments), " 

101 f"got segments={segments!r}" 

102 ) 

103 if not 0 <= brightness <= 7: 

104 raise ValueError(f"Invalid argument: expected 0 <= brightness <= 7, got brightness={brightness!r}") 

105 

106 await self.ipcon.send_request( 

107 device=self, 

108 function_id=FunctionID.SET_SEGMENTS, 

109 data=pack_payload( 

110 ( 

111 list(map(int, segments)), 

112 int(brightness), 

113 bool(colon), 

114 ), 

115 "4B B !", 

116 ), 

117 response_expected=response_expected, 

118 )

120 async def get_segments(self) -> GetSegments: 

121 """ 

122 Returns the segment, brightness and color data as set by 

123 :func:`Set Segments`.

125 _, payload = await self.ipcon.send_request( 

126 device=self, function_id=FunctionID.GET_SEGMENTS, response_expected=True 

127 ) 

128 return GetSegments(*unpack_payload(payload, "4B B !"))

130 async def start_counter( # pylint: disable=too-many-arguments,too-many-positional-arguments 

131 self, 

132 value_from: int, 

133 value_to: int, 

134 increment: int = 1, 

135 length: Duration = Decimal(1), 

136 response_expected: bool = True, 

137 ) -> None: 

138 """ 

139 Starts a counter with the *from* value that counts to the *to* 

140 value each step incremented with by *increment*. 

141 *length* is the pause in s between each increment. 

142 

143 Example: If you set *from* to 0, *to* to 100, *increment* to 1 and 

144 *length* to 1, a counter that goes from 0 to 100 with one second 

145 pause between each increment will be started. 

146 

147 Using a negative increment allows to count backwards. 

148 

149 You can stop the counter at every time by calling :func:`Set Segments`. 

150 """ 

151 if not -999 <= value_from <= 9999: 

152 raise ValueError(f"Invalid argument: expected -999 <= value_from <= 9999, got value_from={value_from!r}") 

153 if not -999 <= value_to <= 9999: 

154 raise ValueError(f"Invalid argument: expected -999 <= value_to <= 9999, got value_to={value_to!r}") 

155 if not -999 <= increment <= 9999: 

156 raise ValueError(f"Invalid argument: expected -999 <= increment <= 9999, got increment={increment!r}") 

157 if not 0 <= seconds_to_ms(length) <= 2**32 - 1: 

158 raise ValueError( 

159 "Invalid argument: expected 0 <= seconds_to_ms(length) <= 2**32 - 1, " 

160 f"got length={length!r}, seconds_to_ms={seconds_to_ms!r}" 

161 ) 

162 

163 await self.ipcon.send_request( 

164 device=self, 

165 function_id=FunctionID.START_COUNTER, 

166 data=pack_payload( 

167 ( 

168 int(value_from), 

169 int(value_to), 

170 int(increment), 

171 seconds_to_ms(length), 

172 ), 

173 "h h h I", 

174 ), 

175 response_expected=response_expected, 

176 )
