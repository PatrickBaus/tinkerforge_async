1""" 

2Conversion between the raw values of the Tinkerforge protocol and SI units. 

3 

4The Bricks and Bricklets transfer physical quantities as integers in a fixed unit, e.g. the temperature in hundredths 

5of a degree Celsius. The library always uses SI units and `Decimal` to preserve the precision. A `Quantity` describes 

6how to convert between both. 

7""" 

8 

9from __future__ import annotations 

10 

11from dataclasses import dataclass 

12from decimal import Decimal 

13 

14CELSIUS_ZERO = Decimal("273.15") 

15"""0 degrees Celsius in K. It is the default of the temperature thresholds, i.e. a raw temperature of 0.""" 

16 

17 

18@dataclass(frozen=True, slots=True) 

19class Quantity: 

20 """ 

21 A physical quantity that is transferred as an integer. The SI value is `(raw + offset) / scale`. 

22 

23 Parameters 

24 ---------- 

25 scale: int 

26 The number of raw units per SI unit, e.g. 1000 if the raw value is in thousandths of a volt. 

27 offset: int 

28 The raw value that is added to the raw value, before it is scaled. It is used for the zero point of the 

29 temperature, e.g. 27315 if the raw value is in hundredths of a degree Celsius and the SI unit is K. 

30 """ 

31 

32 scale: int 

33 offset: int = 0 

34 

35 def to_si(self, raw: int) -> Decimal: 

36 """ 

37 Convert a raw value to SI units. 

38 """ 

39 return Decimal(raw + self.offset) / self.scale 

40 

41 def from_si(self, value: float | Decimal) -> int: 

42 """ 

43 Convert a value in SI units to the raw value. The result is rounded to the nearest integer. The value is 

44 converted to a `Decimal` first, so there are no floating point errors, e.g. 0.0029 * 1e9 = 2899999.9999999995. 

45 """ 

46 return round(Decimal(value) * self.scale) - self.offset 

47 

48 

49TEMPERATURE = Quantity(scale=100, offset=27315) 

50"""The temperature in K. Raw values are in hundredths of a degree Celsius.""" 

51

53@dataclass(frozen=True, slots=True) 

54class LinearQuantity: 

55 """ 

56 A physical quantity that is transferred as an integer, with an arbitrary rational scale. The SI value is 

57 `raw * numerator / denominator + offset`. Unlike `Quantity`, the scale does not need to be the inverse of an 

58 integer, e.g. 1/16 °/s, 9.80665 m/s² per g or 5/18 m/s per km/h. The offset is given in SI units, e.g. 273.15 K for 

59 a temperature, that is transferred in degrees Celsius. 

60 

61 Parameters 

62 ---------- 

63 numerator: int 

64 The numerator of the scale. 

65 denominator: int 

66 The denominator of the scale. 

67 offset: Decimal 

68 The SI value of the raw value 0. 

69 """ 

70 

71 numerator: int 

72 denominator: int = 1 

73 offset: Decimal = Decimal(0) 

74 

75 def to_si(self, raw: int | float) -> Decimal: 

76 """ 

77 Convert a raw value to SI units. 

78 """ 

79 value = Decimal(str(raw)) if isinstance(raw, float) else Decimal(raw) 

80 return value * self.numerator / self.denominator + self.offset 

81 

82 def from_si(self, value: float | Decimal) -> int: 

83 """ 

84 Convert a value in SI units to the raw value. The result is rounded to the nearest integer. 

85 """ 

86 return round((Decimal(str(value)) - self.offset) * self.denominator / self.numerator) 

87 

88 def from_si_float(self, value: float | Decimal) -> float: 

89 """ 

90 Convert a value in SI units to the raw value of a device, that uses floating point numbers. 

91 """ 

92 return float((Decimal(str(value)) - self.offset) * self.denominator / self.numerator)
